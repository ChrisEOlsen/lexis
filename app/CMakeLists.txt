# Qt/C++ desktop UI for LEXIS (see ../dev/APP_SPEC.md). Separate build from the
# top-level Makefile (CLI untouched); lexis_core reads ../src/core/sources.txt.
cmake_minimum_required(VERSION 3.21)
project(lexis_app LANGUAGES C CXX)

# Feeds app/.clangd (per-file Qt-aware flags instead of the root compile_flags.txt).
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

# qtbase and qtdeclarative are keg-only (not brew-linked), so both prefixes must be
# named explicitly. CACHE vars so -D overrides work; defaults match this machine.
set(LEXIS_QTBASE_PREFIX "/opt/homebrew/opt/qtbase" CACHE PATH "qtbase install prefix")
set(LEXIS_QTDECLARATIVE_PREFIX "/opt/homebrew/opt/qtdeclarative" CACHE PATH "qtdeclarative install prefix")
set(CMAKE_PREFIX_PATH "${LEXIS_QTBASE_PREFIX}" "${LEXIS_QTDECLARATIVE_PREFIX}" ${CMAKE_PREFIX_PATH})

# find_package() resolves components relative to Qt6Config.cmake's own prefix, not
# CMAKE_PREFIX_PATH -- point these at qtdeclarative's package configs directly.
set(Qt6Quick_DIR "${LEXIS_QTDECLARATIVE_PREFIX}/lib/cmake/Qt6Quick" CACHE PATH "Qt6Quick package dir")
set(Qt6QuickControls2_DIR "${LEXIS_QTDECLARATIVE_PREFIX}/lib/cmake/Qt6QuickControls2" CACHE PATH "Qt6QuickControls2 package dir")

find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Network)
# REQUIRES 6.5 opts into QTP0001 (resources embed under ":/qt/qml", the prefix the
# engine searches). Without it only the dev build works; an installed .app finds no Main type.
qt_standard_project_setup(REQUIRES 6.5)

# Format extractors: poppler-cpp (PDF), pugixml + libzip (DOCX), tesseract + leptonica
# (OCR). All ship pkg-config files, so no manual path-guessing needed here.
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER_CPP REQUIRED IMPORTED_TARGET poppler-cpp)
pkg_check_modules(PUGIXML REQUIRED IMPORTED_TARGET pugixml)
pkg_check_modules(TESSERACT REQUIRED IMPORTED_TARGET tesseract)
pkg_check_modules(LEPTONICA REQUIRED IMPORTED_TARGET lept)
pkg_check_modules(LIBZIP REQUIRED IMPORTED_TARGET libzip)

set(LEXIS_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

# Single core source list, shared with the top-level Makefile (same file).
# jinja is the one exception: sole C++ unit, listed here with its own flags.
file(STRINGS ${LEXIS_ROOT}/src/core/sources.txt LEXIS_CORE_C_REL)
set(LEXIS_CORE_SRCS "")
foreach(f IN LISTS LEXIS_CORE_C_REL)
    list(APPEND LEXIS_CORE_SRCS ${LEXIS_ROOT}/${f})
endforeach()
list(APPEND LEXIS_CORE_SRCS ${LEXIS_ROOT}/src/core/jinja_chat_template.cpp)

add_library(lexis_core STATIC ${LEXIS_CORE_SRCS})
target_include_directories(lexis_core PUBLIC
    ${LEXIS_ROOT}/include
    ${LEXIS_ROOT}/include/vendor
)
# src/core/vendor holds the minja/nlohmann headers jinja_chat_template.cpp needs.
# PRIVATE: nothing outside that one translation unit needs them.
target_include_directories(lexis_core PRIVATE
    ${LEXIS_ROOT}/src/core/vendor
)

# postgresql@18 is keg-only: resolve includedir/libdir via pg_config itself, not a
# guessed path (pg_config --libdir reports a sibling path, not a subdirectory of opt/).
set(PG_CONFIG /opt/homebrew/opt/postgresql@18/bin/pg_config CACHE FILEPATH "pg_config for postgresql@18")
execute_process(COMMAND ${PG_CONFIG} --includedir OUTPUT_VARIABLE PG_INCLUDE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND ${PG_CONFIG} --libdir OUTPUT_VARIABLE PG_LIB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)

set(LLAMA_CPP_DIR /opt/homebrew/Cellar/llama.cpp/10360 CACHE PATH "llama.cpp Cellar dir")
set(GGML_DIR /opt/homebrew/Cellar/ggml/0.19.0 CACHE PATH "ggml Cellar dir")

# tessdata lives under tesseract's Cellar version path; resolve via brew --prefix
# rather than hardcoding the version number.
execute_process(COMMAND brew --prefix tesseract OUTPUT_VARIABLE TESSERACT_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE)
set(LEXIS_TESSDATA_PREFIX "${TESSERACT_PREFIX}/share/tessdata")

target_include_directories(lexis_core PRIVATE
    ${PG_INCLUDE_DIR}
    ${LLAMA_CPP_DIR}/include
    ${GGML_DIR}/include
)
target_link_directories(lexis_core PUBLIC ${PG_LIB_DIR} ${LLAMA_CPP_DIR}/lib ${GGML_DIR}/lib)
target_link_libraries(lexis_core PUBLIC pq llama ggml ggml-base pthread)

# OFF (default): plain dev binary run from the repo root. ON: LEXIS.app bundle
# for scripts/package_app.sh. Only the output shape changes.
option(LEXIS_BUNDLE "Build lexis_app as a macOS .app bundle for packaging" OFF)

qt_add_executable(lexis_app src/main.cpp)
if(LEXIS_BUNDLE)
    # App icon (generated by scripts/make_icon.py, committed so builds don't need Python).
    set(LEXIS_ICON ${CMAKE_CURRENT_SOURCE_DIR}/assets/LEXIS.icns)
    target_sources(lexis_app PRIVATE ${LEXIS_ICON})
    set_source_files_properties(${LEXIS_ICON} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources")
    set_target_properties(lexis_app PROPERTIES
        MACOSX_BUNDLE TRUE
        OUTPUT_NAME "LEXIS"
        MACOSX_BUNDLE_BUNDLE_NAME "LEXIS"
        MACOSX_BUNDLE_GUI_IDENTIFIER "app.lexis.desktop"
        MACOSX_BUNDLE_BUNDLE_VERSION "1.0"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0"
        MACOSX_BUNDLE_ICON_FILE "LEXIS.icns"
    )
endif()

# QML module: QML_ELEMENT types for `import Lexis`, qml/*.qml compiled in (no .qrc needed).
# QT_QML_SINGLETON_TYPE (must precede QML_FILES): without it Theme.* is undefined at runtime.
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)

qt_add_qml_module(lexis_app
    URI Lexis
    VERSION 1.0
    QML_FILES
        qml/Theme.qml
        qml/Main.qml
        qml/GroupSidebar.qml
        qml/GroupsListView.qml
        qml/GroupDocumentsView.qml
        qml/ChatPanel.qml
        qml/MessageDelegate.qml
        qml/ChatComposer.qml
        qml/ChatHistoryDrawer.qml
        qml/SourceInspectorDialog.qml
        qml/DocumentViewerDialog.qml
        qml/SettingsDialog.qml
    SOURCES
        src/AppController.h
        src/AppController.cpp
        src/ConfigManager.h
        src/ConfigManager.cpp
        src/CorpusListModel.h
        src/CorpusListModel.cpp
        src/DocumentListModel.h
        src/DocumentListModel.cpp
        src/ChatMessageListModel.h
        src/ChatMessageListModel.cpp
        src/ChatSessionListModel.h
        src/ChatSessionListModel.cpp
        src/LexisEngine.h
        src/LexisEngine.cpp
        src/IngestWorker.h
        src/IngestWorker.cpp
        src/ModelLoader.h
        src/ModelLoader.cpp
        src/QueryWorker.h
        src/QueryWorker.cpp
        src/DocxExtractor.h
        src/DocxExtractor.cpp
        src/PdfExtractor.h
        src/PdfExtractor.cpp
        src/OcrExtractor.h
        src/OcrExtractor.cpp
        src/AppEnvironment.h
        src/AppEnvironment.cpp
        src/PostgresManager.h
        src/PostgresManager.cpp
        src/SetupController.h
        src/SetupController.cpp
)
# The generated type-registration file includes QML_ELEMENT headers via <...>;
# src/ must be on the include path or they silently no-op ("undeclared identifier").
target_include_directories(lexis_app PRIVATE ${PG_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(lexis_app PRIVATE LEXIS_TESSDATA_PREFIX="${LEXIS_TESSDATA_PREFIX}")
target_link_libraries(lexis_app PRIVATE lexis_core Qt6::Quick Qt6::QuickControls2 Qt6::Network
    PkgConfig::POPPLER_CPP PkgConfig::PUGIXML PkgConfig::TESSERACT PkgConfig::LEPTONICA PkgConfig::LIBZIP)

# Extraction verification tool (see extract_main.cpp).
qt_add_executable(lexis_extract
    src/extract_main.cpp
    src/DocxExtractor.cpp
    src/PdfExtractor.cpp
    src/OcrExtractor.cpp
)
target_include_directories(lexis_extract PRIVATE ${PG_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(lexis_extract PRIVATE LEXIS_TESSDATA_PREFIX="${LEXIS_TESSDATA_PREFIX}")
target_link_libraries(lexis_extract PRIVATE lexis_core Qt6::Core
    PkgConfig::POPPLER_CPP PkgConfig::PUGIXML PkgConfig::TESSERACT PkgConfig::LEPTONICA PkgConfig::LIBZIP)
set_target_properties(lexis_extract PROPERTIES
    BUILD_RPATH "${LLAMA_CPP_DIR}/lib;${GGML_DIR}/lib"
)

# Batch evaluation harness. Links QueryWorker so runs exercise the exact UI pipeline.
# Qt6::Core only: no QML, no widgets, nothing graphical.
qt_add_executable(lexis_eval
    src/eval_main.cpp
    src/QueryWorker.h
    src/QueryWorker.cpp
)
target_include_directories(lexis_eval PRIVATE ${PG_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(lexis_eval PRIVATE lexis_core Qt6::Core)
set_target_properties(lexis_eval PROPERTIES
    BUILD_RPATH "${LLAMA_CPP_DIR}/lib;${GGML_DIR}/lib"
)

# BUILD_RPATH so the dev binary runs uninstalled (the linker path alone doesn't
# help the dynamic loader at run time).
set_target_properties(lexis_app PROPERTIES
    BUILD_RPATH "${LLAMA_CPP_DIR}/lib;${GGML_DIR}/lib"
)
